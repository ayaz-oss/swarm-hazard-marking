**Goal:** Add the per-robot-map controller C5 and detection noise, run the exploratory experiments E6 and E7 from a frozen copy of the code, close the deferred test gaps, and extend the analysis to E1–E7.

**Architecture:** A batch runner checks out one commit into a git worktree and runs experiments from it, so commits made meanwhile cannot affect a batch. - [ ] **Step 5: Create `scripts/run_batch.py`**

```python
"""Run main experiments from a frozen copy of one commit, so later edits cannot touch them.

Checks out HEAD into .batch/<sha>/ (a git worktree), runs the golden test there, then
runs each experiment from that copy with its src/ first on the import path, writing
results to this checkout's results/. Keeps Windows from idle-sleeping meanwhile and
removes the worktree when everything succeeded.

Usage: python scripts/run_batch.py E6 E7 [--workers 4]
"""

from __future__ import annotations

import argparse
import ctypes
import os
import subprocess
import sys
import time
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]


def git(*args: str, cwd: Path = ROOT) -> str:
    return subprocess.run(["git", *args], cwd=cwd, capture_output=True, text=True, check=True).stdout.strip()


def batch_env(worktree: Path, base: dict | None = None) -> dict:
    env = dict(os.environ if base is None else base)
    src = str(worktree / "src")
    env["PYTHONPATH"] = src + (os.pathsep + env["PYTHONPATH"] if env.get("PYTHONPATH") else "")
    return env


def keep_awake(on: bool) -> None:
    if sys.platform != "win32":
        return
    es_continuous, es_system_required = 0x80000000, 0x00000001
    ctypes.windll.kernel32.SetThreadExecutionState(es_continuous | (es_system_required if on else 0))


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument("experiments", nargs="+", help="experiment names, e.g. E6 E7")
    parser.add_argument("--workers", type=int, default=4)
    args = parser.parse_args()

    if git("status", "--porcelain"):
        print("Refusing: commit your changes first.", file=sys.stderr)
        return 1
    sha = git("rev-parse", "--short", "HEAD")
    worktree = ROOT / ".batch" / sha
    if not worktree.exists():
        git("worktree", "add", "--detach", str(worktree), sha)
    env = batch_env(worktree)
    results = ROOT / "results"
    results.mkdir(exist_ok=True)

    golden = subprocess.run([sys.executable, "-m", "pytest", "-q", "tests/test_golden.py"],
                            cwd=worktree, env=env)
    if golden.returncode != 0:
        print(f"Refusing: golden test failed at {sha}.", file=sys.stderr)
        return 1

    keep_awake(True)
    try:
        for name in args.experiments:
            start = time.time()
            with open(results / f"{name}.log", "w") as log:
                rc = subprocess.run(
                    [sys.executable, str(worktree / "scripts" / "run_experiment.py"),
                     f"configs/experiments/{name}.yaml", "--workers", str(args.workers),
                     "--results-dir", str(results)],
                    cwd=worktree, env=env, stdout=log, stderr=subprocess.STDOUT).returncode
            print(f"{name}: exit {rc}, {time.time() - start:.0f} s (code {sha})", flush=True)
            if rc != 0:
                return rc
    finally:
        keep_awake(False)
    git("worktree", "remove", str(worktree))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

def wreck_repulsion(pos: np.ndarray, wreck_pos: np.ndarray, sense_range: float,
                    detected: np.ndarray | None = None) -> np.ndarray:
    """Sum of pushes away from each visible wreck, stronger when closer.

    `detected[i, j]` False means robot i missed wreck j this step (E7 noise).
    """
    if len(wreck_pos) == 0:
        return np.zeros_like(pos)
    diff = pos[:, None, :] - wreck_pos[None, :, :]
    dist = norm(diff)
    visible = (dist < sense_range) & (dist > 1e-9)
    if detected is not None:
        visible &= detected
    weight = np.where(visible, 1.0 - dist / sense_range, 0.0)
    direction = diff / np.maximum(dist, 1e-9)[..., None]
    return (weight[..., None] * direction).sum(axis=1)
```

    def bump(self, xy, amount: float, sigma: float):
        """The Gaussian bump `deposit` would add: (x slice, y slice, values), or None if off the grid."""
        x, y = float(xy[0]), float(xy[1])
        reach = 3 * sigma
        ix0 = max(int((x - reach) / self.cell), 0)
        ix1 = min(int((x + reach) / self.cell) + 1, self.n)
        iy0 = max(int((y - reach) / self.cell), 0)
        iy1 = min(int((y + reach) / self.cell) + 1, self.n)
        if ix0 >= ix1 or iy0 >= iy1:
            return None
        dx = self.centres[ix0:ix1, None] - x
        dy = self.centres[None, iy0:iy1] - y
        return slice(ix0, ix1), slice(iy0, iy1), amount * np.exp(-(dx**2 + dy**2) / (2 * sigma**2))

    def deposit(self, xy, amount: float, sigma: float) -> None:
        """Add a Gaussian bump with peak `amount`, cut off at 3 sigma."""
        b = self.bump(xy, amount, sigma)
        if b is not None:
            sx, sy, values = b
            self.grid[sx, sy] += values
```

def personal_value(maps: np.ndarray, pts: np.ndarray, cell: float) -> np.ndarray:
    """maps[i] read at pts[i] (cell under each point; off-arena points read the edge cell)."""
    n = maps.shape[1]
    idx = np.clip((np.asarray(pts, dtype=float) / cell).astype(int), 0, n - 1)
    return maps[np.arange(len(idx)), idx[:, 0], idx[:, 1]]


def personal_gradient(maps: np.ndarray, pts: np.ndarray, h: float, cell: float) -> np.ndarray:
    """Each robot's central-difference gradient of its own map (C5)."""
    pts = np.asarray(pts, dtype=float)
    ex = np.array([h, 0.0])
    ey = np.array([0.0, h])
    gx = (personal_value(maps, pts + ex, cell) - personal_value(maps, pts - ex, cell)) / (2 * h)
    gy = (personal_value(maps, pts + ey, cell) - personal_value(maps, pts - ey, cell)) / (2 * h)

    def _share(self) -> None:
        """Synchronous one-hop exchange: each live robot keeps the cell-wise max of its map
        and the maps its live neighbours held before this exchange."""
        sw = self.swarm
        live = np.flatnonzero(sw.alive)

        pairs = np.argwhere(close)
        if len(pairs) == 0:
            return
        old = self.maps.copy()
        for a, b in live[pairs]:
            np.maximum(self.maps[a], old[b], out=self.maps[a])
            np.maximum(self.maps[b], old[a], out=self.maps[b])
