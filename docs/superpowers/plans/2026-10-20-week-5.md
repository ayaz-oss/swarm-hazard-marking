```python
def chain(q, run, steps=20000, pairs=50):
    rng = np.random.default_rng(1)
    blind = None
    trace = np.empty((steps, pairs), dtype=bool)
    for t in range(steps):
        blind = next_blind(blind, rng.random(pairs), q, run)
        trace[t] = blind
    return trace


def mean_spell(trace):
    spells = []
    for col in trace.T:
        edges = np.flatnonzero(np.diff(np.concatenate(([0], col.astype(int), [0]))))
        spells += list(edges[1::2] - edges[::2])
    return np.mean(spells)

    assert mean_spell(trace) == pytest.approx(run, rel=0.1)
```

```python
"""Correlated detection misses (week-5 addendum section 2).

Each (robot, wreck) pair is a two-state chain: blind or seeing. Blind spells last `run`
steps on average and the pair is blind a fraction q of the time. One uniform draw per pair
per step, the same number of draws as independent misses.
"""

from __future__ import annotations

import numpy as np


def next_blind(blind: np.ndarray | None, draws: np.ndarray, q: float, run: float) -> np.ndarray:
    if blind is None:  # first step: start in the stationary distribution
        return draws < q
    stay = 1.0 - 1.0 / run
    enter = q / (run * (1.0 - q))
    return np.where(blind, draws < stay, draws < enter)
```

```python
ESCAPE_WEIGHT = 10.0  # dominates the unit heading, so an oracle robot inside a hazard heads straight out


def hazard_outward(world: World, pos: np.ndarray) -> np.ndarray:
    """Unit vector from each point's nearest hazard centre towards the point."""
    diff = pos[:, None, :] - world.hazard_xy[None, :, :]
    dist = norm(diff)
    rows = np.arange(len(pos))
    nearest = dist.argmin(axis=1)
    return diff[rows, nearest] / np.maximum(dist[rows, nearest], 1e-9)[:, None]
```

```python
"""C6: sighting records shared between neighbours (week-5 addendum section 1).

A record is (position, observer, step), created exactly when C2 would deposit. A robot's
map is the sum of C2's bumps over the records it knows, each decayed by (1 - decay)^age, so
a robot that knows every record has exactly C2's shared field (up to rounding order). Maps
are never stored as grids; a robot's map is read only where the robot samples it.
"""

from __future__ import annotations

import numpy as np

from swarmhazard.geometry import norm
from swarmhazard.pheromone import RepellentField


class SightingRecords:
    def __init__(self, n_robots: int, field: RepellentField, amount: float, sigma: float):
        self.field, self.amount, self.sigma = field, amount, sigma
        self.xy = np.zeros((0, 2))                      # one row per bump position
        self.window = np.zeros((0, 4), dtype=int)       # ix0, ix1, iy0, iy1 of C2's 3-sigma cut-off
        self.rec_pos = np.zeros(0, dtype=int)           # record -> position row
        self.rec_step = np.zeros(0, dtype=int)          # step at which the record was made
        self.know = np.zeros((n_robots, 0), dtype=bool)  # know[i, r]: robot i knows record r
        self._index: dict = {}

    def add(self, key, xy, observer: int, step: int) -> None:
        p = self._index.get(key)
        if p is None:
            sx, sy, _ = self.field.bump(xy, self.amount, self.sigma)
            p = self._index[key] = len(self.xy)
            self.xy = np.vstack((self.xy, np.asarray(xy, dtype=float)[None, :]))
            self.window = np.vstack((self.window, [[sx.start, sx.stop, sy.start, sy.stop]]))
        self.rec_pos = np.append(self.rec_pos, p)
        self.rec_step = np.append(self.rec_step, step)
        column = np.zeros((len(self.know), 1), dtype=bool)
        column[observer] = True
        self.know = np.hstack((self.know, column))

    def _record_weights(self, now: int, decay: float) -> np.ndarray:
        return (1.0 - decay) ** (now - self.rec_step).astype(float)

    def _position_weights(self, know: np.ndarray, now: int, decay: float) -> np.ndarray:
        """(robots, positions): summed decayed amounts of the records each robot knows."""
        onehot = np.zeros((len(self.rec_pos), len(self.xy)))
        onehot[np.arange(len(self.rec_pos)), self.rec_pos] = 1.0
        return (know * self._record_weights(now, decay)[None, :]) @ onehot

    def _bumps_at(self, pts: np.ndarray) -> np.ndarray:
        """(points, positions): C2's unit bump of each position, read in the cell under each point."""
        f = self.field
        idx = np.clip((np.asarray(pts, dtype=float) / f.cell).astype(int), 0, f.n - 1)
        ix, iy = idx[:, 0:1], idx[:, 1:2]
        w = self.window
        inside = (ix >= w[:, 0]) & (ix < w[:, 1]) & (iy >= w[:, 2]) & (iy < w[:, 3])
        dx = f.centres[idx[:, 0]][:, None] - self.xy[None, :, 0]
        dy = f.centres[idx[:, 1]][:, None] - self.xy[None, :, 1]
        return np.where(inside, self.amount * np.exp(-(dx**2 + dy**2) / (2 * self.sigma**2)), 0.0)

    def gradient(self, pts: np.ndarray, h: float, now: int, decay: float) -> np.ndarray:
        """Robot i's central-difference gradient of its own map at pts[i] (same rule as C2)."""
        pts = np.asarray(pts, dtype=float)
        if len(self.rec_pos) == 0:
            return np.zeros_like(pts)
        weights = self._position_weights(self.know, now, decay)

        def value(p):
            return (weights * self._bumps_at(p)).sum(axis=1)

        ex, ey = np.array([h, 0.0]), np.array([0.0, h])
        gx = (value(pts + ex) - value(pts - ex)) / (2 * h)
        gy = (value(pts + ey) - value(pts - ey)) / (2 * h)
        return np.column_stack((gx, gy))

    def share(self, alive: np.ndarray, pos: np.ndarray, comm_range: float) -> None:
        """Synchronous one-hop exchange: each live robot adds what live neighbours knew before."""
        live = np.flatnonzero(alive)
        if len(live) < 2 or self.know.shape[1] == 0:
            return
        close = norm(pos[live][:, None, :] - pos[live][None, :, :]) < comm_range
        close |= np.eye(len(live), dtype=bool)
        self.know[live] = (close.astype(np.int64) @ self.know[live].astype(np.int64)) > 0

    def mean_map(self, alive: np.ndarray, now: int, decay: float) -> np.ndarray:
        """Mean of the live robots' maps (all robots if none live), as a grid."""
        grid = np.zeros((self.field.n, self.field.n))
        if len(self.rec_pos) == 0:
            return grid
        rows = self.know[alive] if alive.any() else self.know
        amounts = self._position_weights(rows, now, decay).mean(axis=0)
        for p in np.flatnonzero(amounts > 0):
            sx, sy, values = self.field.bump(self.xy[p], self.amount * amounts[p], self.sigma)
            grid[sx, sy] += values
        return grid

    def known_frac(self, alive: np.ndarray) -> float:
        if self.know.shape[1] == 0 or not alive.any():
            return float("nan")
        return float(self.know[alive].mean())
```

```python
def pairwise(df, pairs, metrics, rng, n_boot=N_BOOT) -> pd.DataFrame:
    """Within each cell, x vs y for each (x, y) controller pair; two-sided, paired by seed."""
    rows = []
    for cell in sorted(df["cell"].unique()):
        for x_name, y_name in pairs:
            for metric in metrics:
                x, y = paired(df, metric, x_name, y_name, cell=cell)
                row = compare(x, y, rng, n_boot)
                row.update(kind="pair", cell=int(cell), metric=metric, x=x_name, y=y_name,
                           p=wilcoxon_p(x, y, "two-sided"), friedman_p=float("nan"))
                rows.append(row)
    return pd.DataFrame(rows)


def cross_cell(df, controller, cell_x, cell_y, metrics, rng, n_boot=N_BOOT) -> pd.DataFrame:
    """One controller in two cells of a paired-cell experiment, aligned by seed; two-sided."""
    sub = df[df["controller"] == controller]
    rows = []
    for metric in metrics:
        wide = sub.pivot(index="seed", columns="cell", values=metric)[[cell_x, cell_y]].dropna().sort_index()
        x, y = wide[cell_x].to_numpy(float), wide[cell_y].to_numpy(float)
        row = compare(x, y, rng, n_boot)
        row.update(kind="cross_cell", cell=int(cell_x), cell_y=int(cell_y), metric=metric,
                   x=f"{controller}@{cell_x}", y=f"{controller}@{cell_y}",
                   p=wilcoxon_p(x, y, "two-sided"), friedman_p=float("nan"))
        rows.append(row)
    return pd.DataFrame(rows)


def week5_tests(name, df, rng, n_boot=N_BOOT) -> pd.DataFrame:
    spec = WEEK5[name]
    parts = [exploratory(df, rng, n_boot, metrics=spec["metrics"]).assign(kind="vs_C2")]
    if spec["pairs"]:
        parts.append(pairwise(df, spec["pairs"], spec["metrics"], rng, n_boot))
    for controller, cx, cy, metrics in spec["cross"]:
        parts.append(cross_cell(df, controller, cx, cy, metrics, rng, n_boot))
    out = pd.concat(parts, ignore_index=True)
    out["p_holm"] = holm(out["p"].to_numpy())
    return out


def _marked(series) -> bool:
    return any(float(v) != 0.5 for v in str(series).split(";") if v not in ("", "nan"))


def week5_descriptive(df, rng, n_boot=N_BOOT) -> pd.DataFrame:
    """descriptive() plus marked-run counts and the week-5 columns that exist in df."""
    base = descriptive(df, rng, n_boot)
    extra_cols = [c for c in ("delivered_pre", "lost_pre", "delivered_post", "lost_post",
                              "auc_pre", "auc_old_final", "known_frac") if c in df]
    rows = []
    for (cell, controller), sub in df.groupby(["cell", "controller"]):
        row = {"cell": int(cell), "controller": controller}
        row["marked_runs"] = (int(sum(_marked(s) for s in sub["auc_series"].fillna("")))
                              if "auc_series" in sub else 0)
        idx = rng.integers(0, len(sub), size=(n_boot, len(sub)))
        for col in extra_cols:
            values = sub[col].to_numpy(float)
            if np.isnan(values).all():
                row[f"{col}_median"] = row[f"{col}_lo"] = row[f"{col}_hi"] = float("nan")
                continue
            row[f"{col}_median"] = float(np.nanmedian(values))
            row[f"{col}_lo"], row[f"{col}_hi"] = _ci(np.nanmedian(values[idx], axis=1))
        rows.append(row)
    return base.merge(pd.DataFrame(rows), on=["cell", "controller"])


def identity_checks(name, df) -> pd.DataFrame:
    rows = []
    for controller, cells in IDENTITIES.get(name, []):
        sub = df[df["controller"] == controller]
        ref = sub[sub["cell"] == cells[0]].set_index("seed").sort_index()[list(IDENTITY_COLS)]
        same = all(sub[sub["cell"] == k].set_index("seed").sort_index()[list(IDENTITY_COLS)].equals(ref)
                   for k in cells[1:])
        rows.append({"experiment": name, "controller": controller,
                     "cells": " ".join(map(str, cells)), "identical": bool(same)})
    return pd.DataFrame(rows)
```

```python
Usage: python scripts/analyse_week5.py

import subprocess
import sys
from pathlib import Path

import numpy as np
import pandas as pd

sys.path.insert(0, str(Path(__file__).resolve().parent))
from analyse import _golden_at, _golden_branch_problems, completeness_problems, golden_unchanged  # noqa: E402

from swarmhazard.analysis import BOOT_SEED, identity_checks, week5_descriptive, week5_tests  # noqa: E402

RESULTS = ROOT / "results"
OUT = RESULTS / "analysis" / "week5"
E1_COMMIT = "fe78c66"
WEEK5 = ["E6b", "E7b", "E8"]


def provenance_problems(name: str, df: pd.DataFrame) -> list[str]:
    commits = sorted(set(df["git_commit"].astype(str)))
    if len(commits) != 1 or commits[0].endswith("-dirty") or commits[0] == "unknown":
        return [f"{name}: git_commit values {commits} are not one clean commit"]
    c = commits[0]
    try:
        same = golden_unchanged(_golden_at(E1_COMMIT, ROOT), _golden_at(c, ROOT))
    except subprocess.CalledProcessError:
        same = False
    if not same:
        return [f"{name}: golden cases from {E1_COMMIT} are not reproduced at {c}"]
    print(f"{name}: code {c} (every golden case from {E1_COMMIT} reproduced bit-identically)")
    return _golden_branch_problems(name, df, E1_COMMIT, c, ROOT)


def main() -> int:
    missing = [n for n in WEEK5 if not (RESULTS / f"{n}.csv").exists()]
    if missing:
        print("missing: " + ", ".join(missing), file=sys.stderr)
        return 1
    frames = {n: pd.read_csv(RESULTS / f"{n}.csv") for n in WEEK5}
    problems = [p for n, df in frames.items() for p in provenance_problems(n, df) + completeness_problems(n, df)]
    if problems:
        print("Checks failed, no tables written:", file=sys.stderr)
        for p in problems:
            print("  " + p, file=sys.stderr)
        return 1
    OUT.mkdir(parents=True, exist_ok=True)
    checks = []
    for name, df in frames.items():
        week5_tests(name, df, np.random.default_rng(BOOT_SEED)).to_csv(OUT / f"{name}_tests.csv", index=False)
        week5_descriptive(df, np.random.default_rng(BOOT_SEED)).to_csv(OUT / f"{name}_descriptive.csv", index=False)
        checks.append(identity_checks(name, df))
        print(f"wrote {name} tables")
    ids = pd.concat(checks, ignore_index=True)
    ids.to_csv(OUT / "identity_checks.csv", index=False)
    print(ids.to_string(index=False))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
