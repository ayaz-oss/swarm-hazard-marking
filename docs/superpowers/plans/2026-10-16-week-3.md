**Goal:** Run the pre-registered main experiments E1–E5 on a faster, exactly-unchanged simulator, then analyse them exactly as `docs/hypotheses.md` prescribes and produce the first paper figures.

**Architecture:** A golden test pins the simulator's exact output before any speed change. A fifth random stream adds background ("random") failures for the E2 control. A new `swarmhazard.analysis` module implements the pre-registered statistics; `swarmhazard.figures` draws all figures; thin scripts write tables and images from the result CSVs only.

**Tech Stack:** Python 3.11, numpy 1.26.4, scipy 1.15.1, pandas 3.0.0, matplotlib 3.10.0, PyYAML 6.0.3, pytest 9.0.3.

## Global Constraints

- Units are metres, seconds and radians throughout.
- All randomness goes through generators derived from the one passed into `Simulation`. Never call `np.random.*` module functions or Python's `random` (the analysis bootstrap uses its own `numpy.random.default_rng(20261016)`).
- A run is fully determined by `(Config, seed)`; same inputs give an identical result dict.
- Speed changes must keep `tests/test_golden.py` passing bit for bit. A change that breaks it is dropped, not "fixed" by regenerating the golden file.
- `tests/data/golden.json` is generated once (Task 1) and never regenerated in this plan.
- Every main experiment (E1–E5) uses controllers C0–C4, 100 seeds, and `overrides_file: configs/tuned_params.yaml`.
- Never change a parameter, experiment level, statistical method or the model to make a result look better. Report what the code produces.
- Analysis and figures read result CSVs only; they never re-simulate.
- `results/` is git-ignored; never commit CSVs, logs, tables or figures.
- Commands run from the repository root in Git Bash; the interpreter is `.venv/Scripts/python`.
- Every commit message ends with the trailer `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

import math
import sys
from pathlib import Path

import pytest

from swarmhazard.config import Config
from swarmhazard.seeding import make_rng

ROOT = Path(__file__).resolve().parents[1]

def same(a, b) -> bool:
    if isinstance(a, float) and isinstance(b, float) and math.isnan(a) and math.isnan(b):
        return True
    return a == b


@pytest.mark.parametrize("name", sorted(GOLDEN_CASES))

    result = run(Config(**GOLDEN_CASES[name]), make_rng("golden", 0, 0))
    expected = GOLDEN[name]

    diffs = {k: (result[k], v) for k, v in expected.items() if not same(result[k], v)}
    assert not diffs

```python
"""Small geometry helpers shared by the simulator."""

from __future__ import annotations

import numpy as np


def norm(v: np.ndarray) -> np.ndarray:
    """Euclidean length along the last axis.

    Same arithmetic as np.linalg.norm(v, axis=-1) for real arrays
    (sqrt of the summed squares), without its per-call overhead, so results
    are bit-identical (tests/test_golden.py checks this).
    """
    return np.sqrt(np.add.reduce(v * v, axis=-1))
```

```python
    def _pick_up(self, searching: np.ndarray) -> None:
        # Same rule as before: robots in index order take an item within
        # pickup_radius; a taken item respawns at once and may then be taken by
        # a later robot. Distances are computed once; after the first pickup in
        # a step, later robots re-check against the moved item.
        if len(self.world.item_pos) == 0:
            return
        sw = self.swarm
        idx = np.flatnonzero(searching)
        if len(idx) == 0:
            return
        items = self.world.item_pos
        dist = norm(items[None, :, :] - sw.pos[idx][:, None, :])  # (S, M)
        close = np.flatnonzero(dist.min(axis=1) < self.cfg.pickup_radius)
        if len(close) == 0:
            return
        changed = False
        for row in range(close[0], len(idx)):  # rows before the first close one cannot pick up
            i = idx[row]
            d = norm(items - sw.pos[i]) if changed else dist[row]
            j = int(d.argmin())
            if d[j] < self.cfg.pickup_radius:
                sw.state[i] = RETURN
                self.picked_up += 1
                respawn_item(self.world, j, self.respawn_rng)
                changed = True
```

import numpy as np
import pandas as pd
import pytest

from swarmhazard.analysis import E2_NONE_CELL, compare, confirmatory, holm, paired, wilcoxon_p


def frame(rows):
    return pd.DataFrame(rows, columns=["experiment", "cell", "controller", "seed", "delivered", "robots_lost"])


def test_holm_values():
    assert np.allclose(holm([0.01, 0.04, 0.03]), [0.03, 0.06, 0.06])


def test_paired_aligns_by_seed():
    df = frame([("E1", 0, "C2", 1, 10, 0), ("E1", 0, "C0", 0, 5, 0),
                ("E1", 0, "C2", 0, 7, 0), ("E1", 0, "C0", 1, 3, 0)])
    x, y = paired(df, "delivered", "C2", "C0")
    assert x.tolist() == [7, 10] and y.tolist() == [5, 3]


def test_wilcoxon_detects_a_clear_shift_and_ignores_none():
    rng = np.random.default_rng(1)
    y = rng.normal(40, 5, 30)
    assert wilcoxon_p(y + 1.0, y, "greater") < 0.001
    assert wilcoxon_p(y, y, "greater") == 1.0


def test_compare_is_deterministic_and_brackets_the_estimate():
    rng = np.random.default_rng(2)
    x, y = rng.normal(45, 5, 40), rng.normal(42, 5, 40)
    a = compare(x, y, np.random.default_rng(20261016), n_boot=2000)
    b = compare(x, y, np.random.default_rng(20261016), n_boot=2000)
    assert a == b
    assert a["a12_lo"] <= a["a12"] <= a["a12_hi"]
    assert a["median_diff_lo"] <= a["median_diff"] <= a["median_diff_hi"]


def test_confirmatory_on_synthetic_data():

    for s in range(40):

```python
"""The pre-registered analysis (docs/hypotheses.md), as functions over result tables.

Controllers in a cell share random streams seed by seed, so every comparison is
paired by seed: Wilcoxon signed-rank tests (Pratt zero handling) and a bootstrap
that resamples seeds, keeping each seed's values for all controllers together.
"""

from __future__ import annotations

import warnings

import numpy as np
import pandas as pd
from scipy.stats import friedmanchisquare, wilcoxon

from swarmhazard.stats import a12

N_BOOT = 10_000
BOOT_SEED = 20261016
CONTROLLERS = ["C0", "C1", "C2", "C3", "C4"]
E2_NONE_CELL = 3
H3_BOUND = 0.36


def paired(df: pd.DataFrame, metric: str, a: str, b: str, cell: int = 0):
    """Values of `metric` for controllers a and b in one cell, aligned by seed."""
    sub = df[df["cell"] == cell]
    wide = sub.pivot(index="seed", columns="controller", values=metric)[[a, b]].dropna().sort_index()
    return wide[a].to_numpy(float), wide[b].to_numpy(float)


def wilcoxon_p(x, y, alternative: str) -> float:
    x, y = np.asarray(x, float), np.asarray(y, float)
    if np.all(x == y):
        return 1.0
    return float(wilcoxon(x, y, zero_method="pratt", alternative=alternative, method="auto").pvalue)


def holm(pvals) -> np.ndarray:
    p = np.asarray(pvals, float)
    m = len(p)
    adjusted = np.empty(m)
    running = 0.0
    for rank, i in enumerate(np.argsort(p)):
        running = max(running, (m - rank) * p[i])
        adjusted[i] = min(1.0, running)
    return adjusted


def _ci(samples) -> tuple[float, float]:
    lo, hi = np.percentile(samples, [2.5, 97.5])
    return float(lo), float(hi)


def _a12_boot(x, y, idx, chunk: int = 250) -> np.ndarray:
    out = np.empty(len(idx))
    for s in range(0, len(idx), chunk):
        xs = x[idx[s:s + chunk]][:, :, None]
        ys = y[idx[s:s + chunk]][:, None, :]
        out[s:s + chunk] = (xs > ys).mean(axis=(1, 2)) + 0.5 * (xs == ys).mean(axis=(1, 2))
    return out


def compare(x, y, rng: np.random.Generator, n_boot: int = N_BOOT) -> dict:
    """x vs y, paired by seed: medians, median paired difference, A12, seed-bootstrap CIs."""
    x, y = np.asarray(x, float), np.asarray(y, float)
    idx = rng.integers(0, len(x), size=(n_boot, len(x)))
    d = x - y
    diff_lo, diff_hi = _ci(np.median(d[idx], axis=1))
    a_lo, a_hi = _ci(_a12_boot(x, y, idx))
    return {
        "n": int(len(x)),
        "median_x": float(np.median(x)),
        "median_y": float(np.median(y)),
        "median_diff": float(np.median(d)),
        "median_diff_lo": diff_lo,
        "median_diff_hi": diff_hi,
        "a12": a12(x, y),
        "a12_lo": a_lo,
        "a12_hi": a_hi,
    }


def confirmatory(e1: pd.DataFrame, e2: pd.DataFrame | None, rng: np.random.Generator,
                 n_boot: int = N_BOOT) -> pd.DataFrame:
    rows = []
    for hypothesis, metric, a, b in (("H1", "delivered", "C2", "C0"),
                                     ("H2", "robots_lost", "C1", "C2"),
                                     ("H4", "delivered", "C3", "C2")):
        x, y = paired(e1, metric, a, b)
        row = compare(x, y, rng, n_boot)
        row.update(hypothesis=hypothesis, metric=metric, x=a, y=b, p=wilcoxon_p(x, y, "greater"))
        rows.append(row)
    for row, p in zip(rows, holm([r["p"] for r in rows])):
        row["p_holm"] = float(p)
        row["supported"] = bool(p < 0.05)
    if e2 is not None:
        x, y = paired(e2, "delivered", "C2", "C0", cell=E2_NONE_CELL)
        row = compare(x, y, rng, n_boot)
        row.update(hypothesis="H3", metric="delivered", x="C2", y="C0", p=float("nan"),
                   p_holm=float("nan"), supported=bool(row["a12_lo"] > H3_BOUND))
        rows.append(row)
    return pd.DataFrame(rows)


def _friedman(wide: pd.DataFrame) -> float:
    if wide.shape[1] < 3:
        return float("nan")
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        p = friedmanchisquare(*[wide[c].to_numpy(float) for c in wide.columns]).pvalue
    return float(p)


def exploratory(df: pd.DataFrame, rng: np.random.Generator, n_boot: int = N_BOOT,
                metrics=("delivered", "robots_lost")) -> pd.DataFrame:
    """Per cell and metric: Friedman across controllers, then two-sided Wilcoxon vs C2;
    Holm across every pairwise test in the experiment."""
    rows = []
    for cell, sub in df.groupby("cell"):
        present = [c for c in CONTROLLERS if c in set(sub["controller"])]
        if "C2" not in present:
            continue
        for metric in metrics:
            wide = sub.pivot(index="seed", columns="controller", values=metric)[present].dropna().sort_index()
            friedman_p = _friedman(wide)
            for c in present:
                if c == "C2":
                    continue
                x, y = wide[c].to_numpy(float), wide["C2"].to_numpy(float)
                row = compare(x, y, rng, n_boot)
                row.update(cell=int(cell), metric=metric, x=c, y="C2",
                           p=wilcoxon_p(x, y, "two-sided"), friedman_p=friedman_p)
                rows.append(row)
    out = pd.DataFrame(rows)
    if len(out):
        out["p_holm"] = holm(out["p"].to_numpy())
    return out


def descriptive(df: pd.DataFrame, rng: np.random.Generator, n_boot: int = N_BOOT) -> pd.DataFrame:
    """Per cell and controller: medians with seed-bootstrap CIs and secondary metrics."""
    rows = []
    for (cell, controller), sub in df.groupby(["cell", "controller"]):
        delivered = sub["delivered"].to_numpy(float)
        lost = sub["robots_lost"].to_numpy(float)
        idx = rng.integers(0, len(sub), size=(n_boot, len(sub)))
        d_lo, d_hi = _ci(np.median(delivered[idx], axis=1))
        l_lo, l_hi = _ci(np.median(lost[idx], axis=1))
        total_lost = lost.sum()
        rows.append({
            "cell": int(cell), "controller": controller, "n": int(len(sub)),
            "delivered_median": float(np.median(delivered)), "delivered_lo": d_lo, "delivered_hi": d_hi,
            "lost_median": float(np.median(lost)), "lost_lo": l_lo, "lost_hi": l_hi,
            "items_per_robot_lost": float(delivered.sum() / total_lost) if total_lost > 0 else float("nan"),
            "auc_final_median": float(sub["auc_final"].median()) if "auc_final" in sub else float("nan"),
            "hazard_area": float(sub["hazard_area"].iloc[0]) if "hazard_area" in sub else float("nan"),
        })
    return pd.DataFrame(rows)
```

"""All paper figures, drawn from result CSVs only.

One colour per controller everywhere (Okabe–Ito, colour-blind safe). Sweeps show the
median with an interquartile band; inference lives in the analysis tables, not here.
"""

from __future__ import annotations

from pathlib import Path

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt  # noqa: E402
import numpy as np  # noqa: E402
import pandas as pd  # noqa: E402

CONTROLLERS = ["C0", "C1", "C2", "C3", "C4"]
PALETTE = {"C0": "#999999", "C1": "#E69F00", "C2": "#0072B2", "C3": "#009E73", "C4": "#CC79A7"}
NAMES = {"C0": "C0 ignore", "C1": "C1 avoid wrecks", "C2": "C2 mark",
         "C3": "C3 mark + risk", "C4": "C4 oracle"}
METRICS = {"delivered": "Items delivered", "robots_lost": "Robots lost"}
LEVELS = {
    "E2": ["next to patch", "between patch\nand nest", "far", "no hazard\n(random failures)"],
    "E3": [5, 10, 20, 40, 80],
    "E4": [0.005, 0.02, 0.1, 1.0],
    "E5": [0.5, 1.0, 2.0],
}
XLABEL = {"E3": "Swarm size (robots)", "E4": "Lethality per step inside the hazard",
          "E5": "Hazard radius (m)"}
LOGX = {"E3": True, "E4": True, "E5": False}
METRIC_EVERY_S = 50.0  # metric_every (500 steps) x dt (0.1 s)


def _save(fig, out_dir: Path, stem: str) -> list[Path]:
    out_dir.mkdir(parents=True, exist_ok=True)
    paths = [out_dir / f"{stem}.png", out_dir / f"{stem}.pdf"]
    fig.savefig(paths[0], dpi=200, bbox_inches="tight")
    fig.savefig(paths[1], bbox_inches="tight")
    plt.close(fig)
    return paths


def _boxes(ax, df: pd.DataFrame, metric: str) -> None:
    present = [c for c in CONTROLLERS if c in set(df["controller"])]
    data = [df.loc[df["controller"] == c, metric].to_numpy(float) for c in present]
    bp = ax.boxplot(data, widths=0.6, patch_artist=True, showfliers=False)
    for patch, c in zip(bp["boxes"], present):
        patch.set_facecolor(PALETTE[c])
        patch.set_alpha(0.35)
    for k, (c, values) in enumerate(zip(present, data), start=1):
        jitter = np.linspace(-0.15, 0.15, len(values)) if len(values) > 1 else np.zeros(len(values))
        ax.scatter(k + jitter, values, s=5, color=PALETTE[c], alpha=0.6, linewidths=0)
    ax.set_xticks(range(1, len(present) + 1), present)
    ax.set_ylabel(METRICS[metric])


def fig_e1(df: pd.DataFrame, out_dir: Path) -> list[Path]:
    fig, axes = plt.subplots(1, 2, figsize=(8, 3.2))
    for ax, metric in zip(axes, METRICS):
        _boxes(ax, df, metric)
    fig.suptitle("E1: baseline (100 seeds per controller)")
    return _save(fig, out_dir, "e1_baseline")


def fig_e2(df: pd.DataFrame, out_dir: Path) -> list[Path]:
    levels = LEVELS["E2"]
    fig, axes = plt.subplots(2, len(levels), figsize=(3 * len(levels), 5.6), sharey="row")
    for col, label in enumerate(levels):
        sub = df[df["cell"] == col]
        for row, metric in enumerate(METRICS):
            _boxes(axes[row, col], sub, metric)
            if col > 0:
                axes[row, col].set_ylabel("")
        axes[0, col].set_title(label)
    fig.suptitle("E2: hazard placement")
    return _save(fig, out_dir, "e2_placement")


def fig_sweep(df: pd.DataFrame, name: str, out_dir: Path) -> list[Path]:
    x = np.asarray(LEVELS[name], float)
    fig, axes = plt.subplots(1, 2, figsize=(8, 3.2))
    for ax, metric in zip(axes, METRICS):
        for c in CONTROLLERS:
            sub = df[df["controller"] == c]
            if sub.empty:
                continue
            q = sub.groupby("cell")[metric].quantile([0.25, 0.5, 0.75]).unstack()
            q = q.reindex(range(len(x)))
            ax.plot(x, q[0.5], marker="o", color=PALETTE[c], label=NAMES[c])
            ax.fill_between(x, q[0.25], q[0.75], color=PALETTE[c], alpha=0.15, linewidth=0)
        if LOGX[name]:
            ax.set_xscale("log")
        ax.set_xlabel(XLABEL[name])
        ax.set_ylabel(METRICS[metric])
    axes[1].legend(frameon=False, fontsize=8)
    fig.suptitle(f"{name}: median with interquartile band")
    return _save(fig, out_dir, f"{name.lower()}_sweep")


def fig_auc(df: pd.DataFrame, out_dir: Path) -> list[Path]:
    fig, ax = plt.subplots(figsize=(4.5, 3.2))
    for c in ("C2", "C3"):
        series = [s for s in df.loc[df["controller"] == c, "auc_series"].dropna() if str(s)]
        if not series:
            continue
        values = np.array([[float(v) for v in str(s).split(";")] for s in series])
        t = METRIC_EVERY_S * np.arange(1, values.shape[1] + 1)
        ax.plot(t, np.median(values, axis=0), color=PALETTE[c], label=NAMES[c])
        ax.fill_between(t, np.percentile(values, 25, axis=0), np.percentile(values, 75, axis=0),
                        color=PALETTE[c], alpha=0.15, linewidth=0)
    ax.axhline(0.5, color="#666666", linewidth=0.8, linestyle=":")
    ax.set_xlabel("Time (s)")
    ax.set_ylabel("Localisation AUC")
    ax.legend(frameon=False, fontsize=8)
    ax.set_title("E1: how well the repellent map outlines the hazard")
    return _save(fig, out_dir, "e1_auc_over_time")


def make_all(results_dir: Path, out_dir: Path) -> list[Path]:
    written: list[Path] = []
    path = results_dir / "E1.csv"
    if path.exists():
        e1 = pd.read_csv(path)
        written += fig_e1(e1, out_dir)
        written += fig_auc(e1, out_dir)
    path = results_dir / "E2.csv"
    if path.exists():
        written += fig_e2(pd.read_csv(path), out_dir)
    for name in ("E3", "E4", "E5"):
        path = results_dir / f"{name}.csv"
        if path.exists():
            written += fig_sweep(pd.read_csv(path), name, out_dir)
    return written

"""Run the pre-registered analysis (docs/hypotheses.md) on results/E*.csv.

Writes results/analysis/confirmatory.csv and, per experiment, *_descriptive.csv and
*_exploratory.csv. One bootstrap generator (seed 20261016) is used in a fixed order,
so re-running gives identical tables.

Usage: python scripts/analyse.py
"""

from __future__ import annotations

import sys
from pathlib import Path

import numpy as np
import pandas as pd

from swarmhazard.analysis import BOOT_SEED, confirmatory, descriptive, exploratory

ROOT = Path(__file__).resolve().parents[1]
RESULTS = ROOT / "results"
OUT = RESULTS / "analysis"
MAIN = ["E1", "E2", "E3", "E4", "E5"]


def main() -> int:
    frames = {n: pd.read_csv(RESULTS / f"{n}.csv") for n in MAIN if (RESULTS / f"{n}.csv").exists()}
    if "E1" not in frames:
        print("results/E1.csv not found", file=sys.stderr)
        return 1
    for name, df in frames.items():
        commits = sorted(set(df["git_commit"].astype(str)))
        if len(commits) != 1 or commits[0].endswith("-dirty"):
            print(f"WARNING {name}: git_commit values {commits}", file=sys.stderr)
    OUT.mkdir(parents=True, exist_ok=True)
    rng = np.random.default_rng(BOOT_SEED)

    conf = confirmatory(frames["E1"], frames.get("E2"), rng)
    conf.to_csv(OUT / "confirmatory.csv", index=False)
    pd.set_option("display.width", 200)
    print(conf[["hypothesis", "x", "y", "metric", "median_x", "median_y", "median_diff",
                "a12", "a12_lo", "a12_hi", "p", "p_holm", "supported"]].round(4).to_string(index=False))

    for name, df in frames.items():
        descriptive(df, rng).to_csv(OUT / f"{name}_descriptive.csv", index=False)
        exploratory(df, rng).to_csv(OUT / f"{name}_exploratory.csv", index=False)
        print(f"wrote {name} tables")
    return 0


if __name__ == "__main__":
    sys.exit(main())
